```
│   ├── quality.ts         # 质量评分（~30行）
│   ├── aiCost.ts          # AI成本提示（~30行）
│   ├── aiRetry.ts         # AI重试（~20行）

└── en-US/
    ├── index.ts
    ├── common.ts
    └── ...（结构同zh-CN）
```

- 最大模块文件：`cleaning.ts`（~100行）
