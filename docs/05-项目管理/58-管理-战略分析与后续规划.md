```typescript
const sampleSize = aiMode === 'api' 
    ? Math.min(totalRows, 1000)  // API严格

function calculateLocalSampleSize(total: number): number {
    if (total < 5000) return total;      // 全量
```

#### 4. **AI并发优化**（3-5天）
**目标**: 假设生成加速60%

**方案1: 并行生成**
```typescript
// 当前：串行
for (const hypothesis of hypotheses) {
    await generateInsight(hypothesis);
}

// 优化：并行（控制并发3个）
const chunks = chunk(hypotheses, 3);
for (const batch of chunks) {
    await Promise.all(batch.map(h => generateInsight(h)));
}
```

**方案2: 流式返回**
```typescript
// SSE流式推理
const stream = await ai.streamGenerate(prompt);
for await (const chunk of stream) {
    updateUI(chunk);  // 实时显示
}
```
