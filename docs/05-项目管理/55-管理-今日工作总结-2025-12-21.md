### 3. 任务2：AI预加载状态指示+持久化（85%）

**已完成部分（数据层100%）**:

#### localStorage持久化
```typescript
// 初始化：从localStorage恢复
const [fileAIStatus, setFileAIStatus] = useState(() => {
    try {
        const stored = localStorage.getItem('fileAIStatus');
        if (stored) {
            const parsed = JSON.parse(stored);
            const map = new Map(Object.entries(parsed));
            logger.log('AI服务', `从localStorage恢复AI状态: ${map.size}个文件`);
            return map;
        }
    } catch (err) {
        logger.warn('AI服务', 'localStorage读取失败', err);
    }
    return new Map();
});

// 持久化：状态变化时自动写入
useEffect(() => {
    if (fileAIStatus.size > 0) {
        try {
            const obj = Object.fromEntries(fileAIStatus);
            localStorage.setItem('fileAIStatus', JSON.stringify(obj));
        } catch (err) {
            logger.warn('AI服务', 'localStorage写入失败', err);
        }
    }
}, [fileAIStatus]);
```

#### 状态更新逻辑
```typescript
// AI预加载完成后设置状态
setFileAIStatus(prev => {
    const newMap = new Map(prev);
    newMap.set(file.id, { status: 'ready' });
    console.log(`[状态更新] ${file.data.fileName} (${file.id}) → ready`);
    return newMap;
});
```

#### UI渲染逻辑（双key兼容）

}

}
