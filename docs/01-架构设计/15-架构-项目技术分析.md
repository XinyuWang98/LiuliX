### 2.2 并行流水线架构（拆分后）
```typescript
// 主Hook: useFilePreprocessor.ts (<200行)
export const useFilePreprocessor = (projectId: string) => {
  const { 开始质量检查 } = useQualityCheck();
  const { 开始清洗建议缓存 } = useCleaningCache();
  const { 开始洞察假设缓存 } = useInsightCache();
  
  const 开始分析任务 = async (file: DataFile) => {
    // 协调三个子Hook并行执行
    await Promise.allSettled([
      开始质量检查(file),
      开始清洗建议缓存(file),
      开始洞察假设缓存(file)
    ]);
  };
  
  return { 开始分析任务 };
};

// 子Hook示例: useQualityCheck.ts
export const useQualityCheck = () => {
  const { updateFileAnalysis } = useProjectContext();
  
  const 开始质量检查 = async (file: DataFile) => {
    const result = await DataQualityService.assessQuality(file.data);
    updateFileAnalysis(file.id, { quality: result });
  };
  
  return { 开始质量检查 };
};
```

### 2.3 AI成本提示组件设计
```typescript
// AICostWarningModal.tsx
interface AICostWarningProps {
  文件数量: number;
  预计调用次数: number;
  当前剩余额度: number; // Mock值，暂时固定返回999
  onConfirm: () => void;
  onCancel: () => void;
  onConfigureAPI: () => void;
}

export const AICostWarningModal: React.FC<AICostWarningProps> = ({
  文件数量,
  预计调用次数,
  当前剩余额度,
  onConfirm,
  onCancel,
  onConfigureAPI
}) => {
  const { t } = useI18n();
  
  // 计算逻辑
  const 是否额度充足 = 当前剩余额度 >= 预计调用次数;
  
  return (
    <div className="模态遮罩">
      <div className="成本提示卡片">
        <h3>{t('aiCost.title')}</h3>
        <p>{t('aiCost.fileCount', { count: 文件数量 })}</p>
        <p>{t('aiCost.estimatedCalls', { calls: 预计调用次数 })}</p>
        
        {!是否额度充足 && (
          <div className="警告区域">
            <AlertTriangle />
            {t('aiCost.quotaInsufficient')}
          </div>
        )}
        
        <div className="按钮组">
          <button onClick={onConfigureAPI} className="次要按钮">
            {t('aiCost.configureAPI')}
          </button>
          <button onClick={onCancel} className="次要按钮">
            {t('common.cancel')}
          </button>
          <button 
            onClick={onConfirm} 
            className={是否额度充足 ? '主要按钮' : '禁用按钮'}
            disabled={!是否额度充足}
          >
            {t('aiCost.confirmProceed')}
          </button>
        </div>
      </div>
    </div>
  );
};
```

**风险1: 状态竞态（Race Condition）**
- **触发条件**: 用户在预处理完成前快速切换文件
- **缓解方案**:
  ```typescript
  // useFilePreprocessor.ts 中必须实现
  const activeTasksRef = useRef<Map<string, AbortController>>(new Map());
  
  const 开始分析任务 = (fileId: string) => {
    // 1. 取消同一文件的旧任务
    activeTasksRef.current.get(fileId)?.abort();
    
    // 2. 创建新的AbortController
    const controller = new AbortController();
    activeTasksRef.current.set(fileId, controller);
    
    // 3. 执行分析（所有fetch传入signal: controller.signal）
  };
  ```
